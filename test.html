// Unique key for this specific test const TEST_STORAGE_KEY = "poddar_bahupad_test_submitted"; // Check on page load if student has already given the test window.addEventListener("DOMContentLoaded", () => { const savedResult = localStorage.getItem(TEST_STORAGE_KEY); if (savedResult) { showLockedScreen(JSON.parse(savedResult)); } else { initTest(); } }); // 1. YOUR 30 QUESTIONS const quizData = [ { q: "1. रैखिक बहुपद के शून्यक की संख्या कितनी होती है?", options: ["2", "1", "3", "0"], ans: 1 }, { q: "2. बहुपद y³ - 2y² - √3y + 1/2 का घात है:", options: ["1/2", "2", "3", "3/2"], ans: 2 }, // ... Paste your remaining questions here up to 30 ... ]; let countdown; function initTest() { // Render Questions const questionsList = document.getElementById("questions-list"); quizData.forEach((item, index) => { let qDiv = document.createElement("div"); qDiv.className = "question-block"; let optHtml = item.options.map((opt, i) => ` `).join(""); qDiv.innerHTML = `
${item.q}
${optHtml}
`; questionsList.appendChild(qDiv); }); // 45 Minutes Countdown Timer let timeLeft = 45 * 60; const timerElem = document.getElementById("timer"); countdown = setInterval(() => { let minutes = Math.floor(timeLeft / 60); let seconds = timeLeft % 60; seconds = seconds < 10 ? '0' + seconds : seconds; timerElem.innerText = `समय शेष: ${minutes}:${seconds}`; if (timeLeft <= 0) { clearInterval(countdown); alert("समय समाप्त हो गया है! आपका टेस्ट स्वतः सबमिट किया जा रहा है।"); submitTest(); } timeLeft--; }, 1000); } // 2. SUBMISSION LOGIC function submitTest() { clearInterval(countdown); let score = 0; quizData.forEach((item, index) => { const selected = document.querySelector(`input[name="q${index}"]:checked`); if (selected && parseInt(selected.value) === item.ans) { score++; } }); const resultData = { score: score, total: quizData.length, date: new Date().toLocaleDateString("hi-IN") }; // Store in browser to lock future attempts localStorage.setItem(TEST_STORAGE_KEY, JSON.stringify(resultData)); showLockedScreen(resultData); } // 3. SHOW RESULT & LOCK SCREEN function showLockedScreen(data) { document.getElementById("quiz-form").style.display = "none"; document.getElementById("timer").style.display = "none"; const resultBox = document.getElementById("result-box"); resultBox.style.display = "block"; document.getElementById("score-text").innerText = `आपका कुल प्राप्तांक: ${data.score} / ${data.total}`; document.getElementById("feedback-text").innerHTML = `

⚠️ आप यह परीक्षा पहले ही दे चुके हैं। पुनः प्रयास (Retest) की अनुमति नहीं है।

सबमिशन तिथि: ${data.date}

`; }
